def monitor_start():

 def worker():
  with (ROOT/'logs/memory_usage.jsonl').open('a') as f:
   while not event.is_set():
    free,total=torch.cuda.mem_get_info();f.write(json.dumps({'utc':stamp(),'gpu_free_bytes':free,'gpu_total_bytes':total,'allocated_bytes':torch.cuda.memory_allocated(),'reserved_bytes':torch.cuda.memory_reserved(),'host_available_bytes':psutil.virtual_memory().available})+'\n');f.flush();event.wait(2)

def set_eval(model,name,scale=0.9):
 model.set_adapter(name)
 for m in model.modules():
  if hasattr(m,'lora_A') and name in m.lora_A:m.set_scale(name,scale)
 model.gradient_checkpointing_disable();model.config.use_cache=True;model.eval();model.requires_grad_(False)

def training_examples(tok,rows):
 results=[]
 for row in rows:
  user=[{'role':'user','content':row['prompt']}]
  pids=tok.apply_chat_template(user,tokenize=True,add_generation_prompt=True)
  full=tok.apply_chat_template(user+[{'role':'assistant','content':row['answer']}],tokenize=True,add_generation_prompt=False)
  assert full[:len(pids)]==pids,(row['id'],'native chat boundary mismatch')
  assert full[-1]==tok.eos_token_id and len(full)>len(pids)
  labels=[-100]*len(pids)+full[len(pids):]
  assert tok.decode(full[len(pids):-1],skip_special_tokens=True).strip()==row['answer'].strip(),row['id']
  assert len(full)<512 and sum(x!=-100 for x in labels)>1
  results.append({'id':row['id'],'ids':full,'labels':labels,'prompt_tokens':len(pids),'assistant_tokens':len(full)-len(pids)})
 return results

def tensors(tok,items):
 n=max(len(x['ids']) for x in items)
 ids=torch.full((len(items),n),tok.pad_token_id,dtype=torch.long,device='cuda');mask=torch.zeros_like(ids);labels=torch.full_like(ids,-100)
 for i,x in enumerate(items):
  length=len(x['ids']);ids[i,:length]=torch.tensor(x['ids'],device='cuda');mask[i,:length]=1;labels[i,:length]=torch.tensor(x['labels'],device='cuda')
  assert labels[i,length-1].item()==tok.eos_token_id
 assert torch.all(labels[mask==0]==-100)
 return ids,mask,labels

def decode(model,tok,jobs,adapter,condition,logdir,batch_id):
 set_eval(model,adapter)
 context=model.disable_adapter() if condition=='off' else nullcontext()
 def header(h):
  h.update(utc=stamp(),condition=condition,batch_id=batch_id)
  with (ROOT/'logs/token_headers.jsonl').open('a') as f:f.write(json.dumps(h)+'\n')
  print(h['call_id'],'INPUT_TOKENS',h['input_tokens'],'OUTPUT_RESERVE',h['max_new_tokens'],flush=True)
 try:
  with context:r=pd.decode_batch(model,tok,jobs,temperature=0.6,do_sample=True,expected_adapter=adapter,expected_adapter_scale=0.9,expected_adapters_enabled=condition!='off',context_limit=16384,prefill_chunk_size=512,header_sink=header)
 except pd.BatchDecodeError as e:
  save(ROOT/'logs/batches'/f'{batch_id}_failed.json',e.partial_result);raise
 finally:model.eval();model.requires_grad_(False)
 save(ROOT/'logs/batches'/f'{batch_id}.json',r['batch'])
 for row,job in zip(r['results'],jobs):
  assert row['call_id']==job['call_id']
  assert row['output']==tok.decode(row['output_token_ids'],skip_special_tokens=True,clean_up_tokenization_spaces=False)
  row.update(condition=condition,adapter=adapter,prompt=job['prompt'],component=job['component'],manager=job['manager'],utc=stamp())
  path=ROOT/logdir/f"{job['call_id']}.json";assert not path.exists();save(path,row)
 return r['results']

def evaluate_validation(model,tok,manager,epoch,rows,key,tokenized):
 name='QA_'+manager
 jobs=[{'call_id':f'{manager}_epoch{epoch:02d}_article{r["article_id"]}_{r["component"]}_validation','manager':manager,'component':r['component'],'prompt':r['prompt'],'prefill':'','seed':seed('phase10_validation_'+manager+'_'+r['article_id']+'_'+r['component']),'max_new_tokens':64} for r in rows]
 results=decode(model,tok,jobs,name,'on','logs/validation',f'{manager}_epoch{epoch:02d}_validation')
 # Generated answers are retained as diagnostics only; no regex-based pass determines selection.
 ids,mask,labels=tensors(tok,tokenized)
 with torch.inference_mode(),torch.autocast('cuda',dtype=torch.bfloat16): loss=float(model(input_ids=ids,attention_mask=mask,labels=labels,use_cache=False).loss)
 del ids,mask,labels
 record={'manager':manager,'epoch':epoch,'total':len(rows),'assistant_validation_loss':loss,'calls':[r['call_id'] for r in results],'utc':stamp()}
 print('VALIDATION',json.dumps(record),flush=True);save(ROOT/'logs/training'/f'{manager}_epoch{epoch:02d}_selection.json',record)
 return record
